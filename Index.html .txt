    <script>
        // Variables globales
        let rawData = [];
        let filteredData = [];
        let columns = [];
        let numCols = [];
        let catCols = [];
        let geoCols = []; // Detecta si hay columnas de provincia, ciudad o país
        let currentFilter = null;

        // Elementos del DOM
        const dropZone = document.getElementById('drop-zone');
        const fileInput = document.getElementById('file-input');
        const loading = document.getElementById('loading');
        const dashboardContent = document.getElementById('dashboard-content');
        const resetFiltersBtn = document.getElementById('reset-filters');

        // Event Listeners Drag & Drop
        dropZone.addEventListener('click', () => fileInput.click());
        dropZone.addEventListener('dragover', (e) => { e.preventDefault(); dropZone.classList.add('drag-active'); });
        dropZone.addEventListener('dragleave', () => dropZone.classList.remove('drag-active'));
        dropZone.addEventListener('drop', (e) => {
            e.preventDefault();
            dropZone.classList.remove('drag-active');
            if (e.dataTransfer.files.length) handleFile(e.dataTransfer.files[0]);
        });
        fileInput.addEventListener('change', (e) => {
            if (e.target.files.length) handleFile(e.target.files[0]);
        });

        resetFiltersBtn.addEventListener('click', () => {
            currentFilter = null;
            filteredData = [...rawData];
            resetFiltersBtn.classList.add('hidden');
            updateDashboard();
        });

        // Procesamiento del archivo Excel
        function handleFile(file) {
            dropZone.style.display = 'none';
            loading.classList.remove('hidden');

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const data = new Uint8Array(e.target.result);
                    const workbook = XLSX.read(data, {type: 'array', cellDates: true});
                    const firstSheetName = workbook.SheetNames[0];
                    const worksheet = workbook.Sheets[firstSheetName];
                    
                    rawData = XLSX.utils.sheet_to_json(worksheet, {defval: ""});
                    filteredData = [...rawData];
                    
                    analyzeStructure();
                    updateDashboard();
                    
                    loading.classList.add('hidden');
                    dashboardContent.classList.remove('hidden');
                } catch (error) {
                    alert("Error leyendo el archivo. Asegúrate de que sea un Excel válido.");
                    console.error(error);
                    location.reload();
                }
            };
            reader.readAsArrayBuffer(file);
        }

        // Analiza inteligentemente las columnas para saber qué graficar
        function analyzeStructure() {
            if(rawData.length === 0) return;
            columns = Object.keys(rawData[0]);
            numCols = [];
            catCols = [];
            geoCols = [];

            columns.forEach(col => {
                const sample = rawData.find(row => row[col] !== "" && row[col] !== null);
                if (!sample) return;
                
                const val = sample[col];
                if (typeof val === 'number') {
                    numCols.push(col);
                } else if (typeof val === 'string') {
                    // Check if it looks like geography
                    const colLower = col.toLowerCase();
                    if(colLower.includes('provincia') || colLower.includes('ciudad') || colLower.includes('pais') || colLower.includes('estado')) {
                        geoCols.push(col);
                    }
                    // Avoid dates categorized as simple strings if possible, but keep it simple
                    if(new Set(rawData.map(r => r[col])).size < rawData.length * 0.8) { 
                        // Only add as categorical if it's not mostly unique values (like IDs)
                        catCols.push(col);
                    }
                }
            });

            // Fallbacks si el algoritmo es muy estricto
            if(catCols.length === 0) catCols = columns.filter(c => !numCols.includes(c));
        }

        // Actualiza todo el Dashboard basado en filteredData
        function updateDashboard() {
            renderKPIs();
            renderCharts();
            renderTable();
            generateConclusions();
        }

        function renderKPIs() {
            const container = document.getElementById('kpi-container');
            container.innerHTML = '';
            
            // KPI 1: Total Registros
            container.appendChild(createKpiCard('Total Registros', filteredData.length.toLocaleString()));

            // Buscar columnas numéricas principales para KPIs
            let numRendered = 1;
            numCols.slice(0, 3).forEach(col => {
                const sum = filteredData.reduce((acc, row) => acc + (Number(row[col]) || 0), 0);
                // Si la suma es muy pequeña o cero constante, quizas es un ID
                if(sum > 0) {
                    let formatted = sum > 1000 ? sum.toLocaleString(undefined, {maximumFractionDigits: 2}) : sum.toFixed(2);
                    container.appendChild(createKpiCard(`Total ${col}`, formatted));
                    numRendered++;
                }
            });

            // Rellenar si faltan KPIs
            if(numRendered < 4 && catCols.length > 0) {
                const uniqueVals = new Set(filteredData.map(r => r[catCols[0]])).size;
                container.appendChild(createKpiCard(`Categorías (${catCols[0]})`, uniqueVals));
            }
        }

        function createKpiCard(title, value) {
            const div = document.createElement('div');
            div.className = 'bg-white p-6 rounded-xl shadow border border-gray-100 flex flex-col justify-center items-center text-center';
            div.innerHTML = `<p class="text-sm text-gray-500 uppercase font-semibold tracking-wider mb-1">${title}</p>
                             <p class="text-3xl font-bold text-blue-600">${value}</p>`;
            return div;
        }

        function renderCharts() {
            if(catCols.length === 0) return;

            const mainCat = catCols[0]; // Usar la primera columna categórica
            const mainNum = numCols.length > 0 ? numCols[0] : null;

            // Agrupar datos para gráficos
            const grouped = {};
            filteredData.forEach(row => {
                const key = row[mainCat] || 'N/A';
                if(!grouped[key]) grouped[key] = { count: 0, sum: 0 };
                grouped[key].count++;
                if(mainNum) grouped[key].sum += (Number(row[mainNum]) || 0);
            });

            // Ordenar por valor
            const sortedKeys = Object.keys(grouped).sort((a, b) => grouped[b].count - grouped[a].count).slice(0, 15); // Top 15
            const labels = sortedKeys;
            const values = sortedKeys.map(k => mainNum ? grouped[k].sum : grouped[k].count);

            // 1. Gráfico de Barras (Permite filtrado)
            const traceBar = {
                x: labels, y: values,
                type: 'bar',
                marker: { color: '#3b82f6' }
            };
            const layoutBar = { 
                margin: {t: 20, l: 40, r: 20, b: 60}, 
                xaxis: { title: mainCat }, 
                yaxis: { title: mainNum ? `Suma de ${mainNum}` : 'Cantidad' }
            };
            Plotly.newPlot('chart-bar', [traceBar], layoutBar, {responsive: true});
            
            // Interacción: Filtrar al pulsar la barra
            document.getElementById('chart-bar').on('plotly_click', function(data){
                const selected = data.points[0].x;
                currentFilter = { col: mainCat, val: selected };
                filteredData = rawData.filter(row => row[mainCat] == selected);
                resetFiltersBtn.classList.remove('hidden');
                updateDashboard();
            });

            // 2. Gráfico Circular (Pie)
            const secCat = catCols.length > 1 ? catCols[1] : mainCat;
            const groupedPie = {};
            filteredData.forEach(row => {
                const key = row[secCat] || 'N/A';
                groupedPie[key] = (groupedPie[key] || 0) + 1;
            });
            const pieLabels = Object.keys(groupedPie).slice(0, 10);
            const pieValues = pieLabels.map(k => groupedPie[k]);

            const tracePie = {
                labels: pieLabels, values: pieValues,
                type: 'pie', hole: 0.4,
                marker: { colors: ['#3b82f6', '#10b981', '#f59e0b', '#ef4444', '#8b5cf6'] }
            };
            Plotly.newPlot('chart-pie', [tracePie], {margin: {t: 20, l: 20, r: 20, b: 20}}, {responsive: true});

            // 3. Gráfico de Líneas / Scatter (o Mapa si hay datos geo)
            const mapLineContainer = document.getElementById('chart-line-map');
            if (numCols.length >= 2) {
                // Scatter / Líneas si hay múltiples números
                const traceScatter = {
                    x: filteredData.map(r => r[numCols[0]]),
                    y: filteredData.map(r => r[numCols[1]]),
                    mode: 'markers', type: 'scatter',
                    marker: { size: 8, color: '#10b981', opacity: 0.6 }
                };
                Plotly.newPlot('chart-line-map', [traceScatter], {
                    margin: {t: 20}, xaxis: {title: numCols[0]}, yaxis: {title: numCols[1]}
                }, {responsive: true});
            } else if (geoCols.length > 0) {
                // Pseudo-mapa (Gráfico de barras horizontal por ubicación)
                const geoAgrupado = {};
                filteredData.forEach(r => {
                    const key = r[geoCols[0]] || 'N/A';
                    geoAgrupado[key] = (geoAgrupado[key] || 0) + 1;
                });
                const locs = Object.keys(geoAgrupado).sort((a,b) => geoAgrupado[a] - geoAgrupado[b]);
                const locVals = locs.map(k => geoAgrupado[k]);
                
                const traceGeo = {
                    y: locs, x: locVals, type: 'bar', orientation: 'h', marker: {color: '#8b5cf6'}
                };
                Plotly.newPlot('chart-line-map', [traceGeo], {margin: {t:20, l: 120}}, {responsive: true});
            } else {
                 mapLineContainer.innerHTML = '<div class="h-full flex items-center justify-center text-gray-400">Datos insuficientes para análisis de tendencias secundarias.</div>';
            }
        }

        function generateConclusions() {
            const list = document.getElementById('conclusions-list');
            list.innerHTML = '';
            
            const li1 = document.createElement('li');
            li1.innerText = `Se están analizando ${filteredData.length} registros.` + (currentFilter ? ` (Filtrado por ${currentFilter.col}: ${currentFilter.val})` : '');
            list.appendChild(li1);

            if(catCols.length > 0 && numCols.length > 0) {
                // Encontrar el mayor contribuyente
                const mainCat = catCols[0];
                const mainNum = numCols[0];
                const grouped = {};
                let totalSum = 0;
                
                filteredData.forEach(row => {
                    const key = row[mainCat] || 'N/A';
                    const val = Number(row[mainNum]) || 0;
                    grouped[key] = (grouped[key] || 0) + val;
                    totalSum += val;
                });

                if (totalSum > 0) {
                    const topCat = Object.keys(grouped).reduce((a, b) => grouped[a] > grouped[b] ? a : b);
                    const percentage = ((grouped[topCat] / totalSum) * 100).toFixed(1);
                    
                    const li2 = document.createElement('li');
                    li2.innerHTML = `La categoría <strong>${topCat}</strong> en la columna <em>${mainCat}</em> representa el <strong>${percentage}%</strong> del valor total de <em>${mainNum}</em>.`;
                    list.appendChild(li2);
                }
            }
        }

        function renderTable() {
            const thead = document.getElementById('table-head');
            const tbody = document.getElementById('table-body');
            thead.innerHTML = ''; tbody.innerHTML = '';

            if (filteredData.length === 0) return;

            // Headers (Mostrar máx 10 columnas para no romper el layout)
            const displayCols = columns.slice(0, 10);
            const trHead = document.createElement('tr');
            displayCols.forEach(col => {
                const th = document.createElement('th');
                th.className = 'px-6 py-3';
                th.innerText = col;
                trHead.appendChild(th);
            });
            thead.appendChild(trHead);

            // Body (Máximo 100 filas)
            const rowsToShow = filteredData.slice(0, 100);
            rowsToShow.forEach(row => {
                const tr = document.createElement('tr');
                tr.className = 'bg-white border-b hover:bg-gray-50';
                displayCols.forEach(col => {
                    const td = document.createElement('td');
                    td.className = 'px-6 py-4 truncate max-w-xs';
                    // Formatear numéricos si es necesario
                    const val = row[col];
                    td.innerText = typeof val === 'number' && !Number.isInteger(val) ? val.toFixed(2) : (val || '-');
                    tr.appendChild(td);
                });
                tbody.appendChild(tr);
            });
        }
    </script>
